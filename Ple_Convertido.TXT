// funcion segura
Func<object, decimal> SafeRead = (obj) =>
{
    if (obj == null) return 0m;

    decimal temp = 0m;
    decimal.TryParse(obj.ToString().Replace(",", "."), out temp);
    return temp;
};

// Creacion de diccionarios para validaciones 02, 03 y 05
var dictClientes = new Dictionary<string, HashSet<string>>();
var dictDocs = new Dictionary<string, HashSet<string>>();
var dictFacturas = new Dictionary<string, decimal>();
var dictNotas = new Dictionary<string, decimal>();

// Llenado de diccionarios para validaciones 02, 03 y 05
foreach (System.Data.DataRow row in DT_PLE_Table.Rows)
{
    string nombre = "";
    string doc = "";
    string tipo = row[idxTipoComp] == DBNull.Value ? "" : row[idxTipoComp].ToString().Trim();
    decimal total = SafeRead(row[idxTotal]);

    // Captura nombre
    if (row[idxNombre] != null)
{
        nombre = row[idxNombre].ToString().Trim().ToUpper();
}

    // Captura documento de identidad
    if (row[idxDoc] != null)
{
        doc = row[idxDoc].ToString().Trim();
}

    if (!dictClientes.ContainsKey(nombre))
{
        dictClientes[nombre] = new HashSet<string>();
}

    if (!dictDocs.ContainsKey(doc))
{
        dictDocs[doc] = new HashSet<string>();
}

    dictClientes[nombre].Add(doc);
    dictDocs[doc].Add(nombre);

    // Facturas
    if (tipo == "1")
{
        string key =
(row[idxTipoComp] == DBNull.Value ? "" : row[idxTipoComp].ToString().Trim()) +
(row[idxSerie] == DBNull.Value ? "" : row[idxSerie].ToString().Trim()) +
(row[idxNumero] == DBNull.Value ? "" : row[idxNumero].ToString().Trim()) +
(row[idxCampoFinal] == DBNull.Value ? "" : row[idxCampoFinal].ToString().Trim());

        if (!dictFacturas.ContainsKey(key))
            dictFacturas[key] = 0;

        dictFacturas[key] += total;
}

    // NC (Nota de credito) / ND (Nota de debito)
    if (tipo == "7" || tipo == "8")
{
        string key =
(row[idxTipoMod] == DBNull.Value ? "" : row[idxTipoMod].ToString().Trim()) +
(row[idxSerieMod] == DBNull.Value ? "" : row[idxSerieMod].ToString().Trim()) +
(row[idxNumeroMod] == DBNull.Value ? "" : row[idxNumeroMod].ToString().Trim());

        if (!dictNotas.ContainsKey(key))
            dictNotas[key] = 0;

        dictNotas[key] += total;
}
}

foreach (System.Data.DataRow row in DT_PLE_Table.Rows)
{
    decimal valExport = SafeRead(row[idxExport]);
    decimal valExonerada = SafeRead(row[idxExonerada]);
    decimal valInafecta = SafeRead(row[idxInafecta]);
    decimal valIGV = SafeRead(row[idxIGV]);
    decimal valDescIGV = SafeRead(row[idxDescIGV]);
    decimal baseGravada = SafeRead(row[idxBaseGravada]);
    decimal descBase = SafeRead(row[idxDescBase]);
    decimal isc = SafeRead(row[idxISC]);
    decimal baseArroz = SafeRead(row[idxBaseArroz]);
    decimal igvArroz = SafeRead(row[idxIGVArroz]);
    decimal impBolsas = SafeRead(row[idxImpBolsas]);
    decimal otros = SafeRead(row[idxOtros]);
    decimal totalComprobante = SafeRead(row[idxTotal]);

    string tipoOperacion = "";
    string nombre = row[idxNombre] == DBNull.Value ? "" : row[idxNombre].ToString().Trim().ToUpper();
    string tipo = row[idxTipoComp] == DBNull.Value ? "" : row[idxTipoComp].ToString().Trim();
    string doc = row[idxDoc] == DBNull.Value ? "" : row[idxDoc].ToString().Trim();
    string obs = row[idxObs] == DBNull.Value ? "" : row[idxObs].ToString();

    // ------------------------
    // VALIDACION 01: Tipo de Operación
    // ------------------------
    if (valExport > 0)
{
        tipoOperacion = "Exportación";
}
    else if ((valInafecta > 0 || valExonerada > 0) && valIGV == 0 && valDescIGV == 0)
{
        tipoOperacion = "Posible Exportación";
}
    else if (valIGV > 0 || valDescIGV > 0)
{
        tipoOperacion = "Operaciones Gravadas";
}
    else
{
        tipoOperacion = "Operaciones Anuladas";
}

    // ------------------------
    // VALIDACION 02: Nombre → múltiples documentos
    // ------------------------
    if (!string.IsNullOrEmpty(nombre) 
&& dictClientes.ContainsKey(nombre) 
&& dictClientes[nombre].Count > 1)
{
        string nuevaObs = "El huésped presenta varios registros con distintos documentos de identidad.";

        if (string.IsNullOrEmpty(obs))
            obs = nuevaObs;
        else if (!obs.Contains(nuevaObs))
            obs = obs + " " + nuevaObs;
}

    // ------------------------
    // VALIDACION 03: Documento → múltiples nombres
    // ------------------------
    if (!string.IsNullOrEmpty(doc) 
&& dictDocs.ContainsKey(doc) 
&& dictDocs[doc].Count > 1)
{
        string nuevaObs = "El documento de identidad está asociado a huéspedes diferentes.";

        if (string.IsNullOrEmpty(obs))
            obs = nuevaObs;
        else if (!obs.Contains(nuevaObs))
            obs = obs + " " + nuevaObs;
}

    // ------------------------
    // VALIDACION 04: Compara Suma de Importes con Importe Total
    // ------------------------
    decimal sumaImportes = valExport + baseGravada + descBase + valIGV + valDescIGV + valExonerada + valInafecta + isc + baseArroz + igvArroz + impBolsas + otros;
    if (Math.Abs(sumaImportes - totalComprobante) > 0.01m) // Considerando un margen de error de 0.01
{
        string nuevaObs = "La suma de los importes no coincide con el importe total del comprobante.";

        if (string.IsNullOrEmpty(obs))
            obs = nuevaObs;
        else if (!obs.Contains(nuevaObs))
            obs = obs + " " + nuevaObs;

}

    // ------------------------
    // VALIDACION 05:
    // ------------------------
    if (tipo == "7" || tipo == "8" || tipo == "1")
{
        string keyRef = "";

        if (tipo == "1")
{
            keyRef =
(row[idxTipoComp] == DBNull.Value ? "" : row[idxTipoComp].ToString().Trim()) +
(row[idxSerie] == DBNull.Value ? "" : row[idxSerie].ToString().Trim()) +
(row[idxNumero] == DBNull.Value ? "" : row[idxNumero].ToString().Trim()) +
(row[idxCampoFinal] == DBNull.Value ? "" : row[idxCampoFinal].ToString().Trim());
}
        else
{
            keyRef =
(row[idxTipoMod] == DBNull.Value ? "" : row[idxTipoMod].ToString().Trim()) +
(row[idxSerieMod] == DBNull.Value ? "" : row[idxSerieMod].ToString().Trim()) +
(row[idxNumeroMod] == DBNull.Value ? "" : row[idxNumeroMod].ToString().Trim());
}

        if (dictFacturas.ContainsKey(keyRef) && dictNotas.ContainsKey(keyRef))
{
            decimal sumaTotal = dictFacturas[keyRef] + dictNotas[keyRef];

            if (sumaTotal <= 0)
{
                string nuevaObs = "Dos notas de crédito o más anulan el importe total de un mismo comprobante.";

                if (string.IsNullOrEmpty(obs))
                    obs = nuevaObs;
                else if (!obs.Contains(nuevaObs))
                    obs = obs + " " + nuevaObs;
}
}
}

    // Si no hay observaciones, asignar mensaje de validación
    if (string.IsNullOrWhiteSpace(obs))
        obs = "OK";

    // ------------------------
    // GUARDAR RESULTADOS
    // ------------------------
    row[idxTipoOperacion] = tipoOperacion;
    row[idxObs] = obs;
}
